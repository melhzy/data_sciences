% preamble.tex — LaTeX preamble for GRAD 695 syllabus

% ----- Fonts & encoding -------------------------------------------------------
\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}

% ----- Page geometry ----------------------------------------------------------
\usepackage{geometry}
\geometry{margin=1in, top=1.2in, bottom=1in}

% ----- Colours ----------------------------------------------------------------
\usepackage{xcolor}
\definecolor{huBlue}{RGB}{0, 51, 102}
\definecolor{boxbg}{RGB}{240, 245, 255}
\definecolor{boxborder}{RGB}{0, 51, 102}

% ----- Hyperlinks -------------------------------------------------------------
\usepackage{hyperref}
\hypersetup{
  colorlinks = true,
  urlcolor   = huBlue,
  linkcolor  = huBlue,
  citecolor  = huBlue
}

% ----- Graphics --------------------------------------------------------------
\usepackage{graphicx}

% ----- Tables -----------------------------------------------------------------
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{array}

% ----- Coloured callout boxes (blackbox environment used in schedule) ---------
\usepackage{framed}

\newenvironment{blackbox}{%
  \def\FrameCommand{%
    \hspace{1pt}%
    {\color{boxborder}\vrule width 2pt}%
    {\color{boxbg}\vrule width 4pt}%
    \colorbox{boxbg}%
  }%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \noindent\ignorespaces
}{%
  \endMakeFramed
}

% ----- Section spacing --------------------------------------------------------
\usepackage{titlesec}
\titlespacing*{\section}{0pt}{2ex plus 1ex minus .2ex}{1ex plus .2ex}
\titlespacing*{\subsection}{0pt}{1.5ex plus 1ex minus .2ex}{0.5ex plus .2ex}
